    86→    86→function todayISO(): string {
    87→    87→  // Local date in Asia/Jakarta — kept simple, avoids TZ drift in input[type=date]
    88→    88→  const now = new Date();
    89→    89→  const tz = "Asia/Jakarta";
    90→    90→  const fmt = new Intl.DateTimeFormat("sv-SE", {
    91→    91→    timeZone: tz,
    92→    92→    year: "numeric",
    93→    93→    month: "2-digit",
    94→    94→    day: "2-digit",
    95→    95→  });
    96→    96→  return fmt.format(now); // YYYY-MM-DD
    97→    97→}

    99→    99→function formatDisplayDate(iso: string): string {
   100→   100→  if (!iso) return "-";
   101→   101→  const [y, m, d] = iso.split("-").map(Number);
   102→   102→  if (!y || !m || !d) return iso;
   103→   103→  const date = new Date(y, m - 1, d);
   104→   104→  return new Intl.DateTimeFormat("id-ID", {
   105→   105→    day: "2-digit",
   106→   106→    month: "short",
   107→   107→    year: "numeric",
   108→   108→  }).format(date);
   109→   109→}
